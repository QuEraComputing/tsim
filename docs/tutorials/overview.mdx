title: "Overview"

Tsim is a quantum circuit sampler that can efficiently sample from Clifford+T circuits with Pauli noise. It is based on ZX-calculus stabilizer rank decomposition and parametrized ZX diagrams, following work of [arXiv:2403.06777](https://arxiv.org/abs/2403.06777).

## Supported Gates

Tsim supports a universal gate set, together with measurement and reset instructions, and Pauli noise channels.

### Clifford Instructions

Tsim supports all instructions [supported by Stim](https://github.com/quantumlib/Stim/wiki/Stim-v1.13-Gate-Reference). Below, we show the standard generating set of Clifford gates:


```python
from tsim import Circuit

c = Circuit("H 0")
c.diagram("timeline-svg", height=120)
```

(function() {
  var wrap = document.querySelector('[data-tsim-zoom=&quot;tsim-zoom-001&quot;]');

    size.style.height = (natH * scale) + 'px';
  }
  apply();
  wrap.addEventListener('wheel', function(e) {
    if (e.ctrlKey || e.metaKey) {
      e.preventDefault();
      var rect = wrap.getBoundingClientRect();
      var mx = e.clientX - rect.left + wrap.scrollLeft;
      var my = e.clientY - rect.top + wrap.scrollTop;
      var factor = Math.exp(-e.deltaY * 0.01);
      var newScale = Math.min(Math.max(0.02, scale * factor), 40);
      var ratio = newScale / scale;
      scale = newScale;
      apply();
      wrap.scrollLeft = mx * ratio - (e.clientX - rect.left);
      wrap.scrollTop = my * ratio - (e.clientY - rect.top);
    }
  }, { passive: false });
})();

Computation time and memory requirement scales exponentially with the number of non-Clifford gates.

The `SPP` instruction applies $e^{-i \pi/4 \cdot P}$ (up to global phase) for a Pauli product $P$, which is equivalent to phasing the $-1$ eigenspace of $P$ by $i$. The `!` operator inverts the product, swapping `SPP` and `SPP_DAG` behavior.

    CNOT 0 1
    M 0 1

sampler = c.compile_sampler()

The second sampling mode is detector sampling. This will sample detector events and observable values. Detector and observable bits can always be obtained by linear transformations of the measurement bits as return by the measurement sampler.
In practice, however, it can be much more efficient to sample detector events directly.


```python
c = Circuit("""
    RX 0
    R 1
    CNOT 0 1
    M 0 1
    DETECTOR rec[-1] rec[-2]
    OBSERVABLE_INCLUDE(0) rec[-1]
""")
sampler = c.compile_detector_sampler()
c.diagram("timeline-svg", height=170)
```

    // &quot;(e1⊕f0)π&quot; -&gt; &quot;e1⊕f0&quot;).
    function displayedPhaseTerm(term) {
        if (!term || !term.endsWith(&quot;π&quot;)) {
            return term;
        }
        const withoutPi = term.slice(0, -1);
        if (/^~?\d*$/.test(withoutPi)) {
            return term;
        }
        if (withoutPi.startsWith(&quot;(&quot;) &amp;&amp; withoutPi.endsWith(&quot;)&quot;)) {
            return withoutPi.slice(1, -1);
        }
        return withoutPi;
    }

    function displayedPhaseText(phase, expanded) {
        return (expanded ? phase : collapsedPhaseText(phase))
            .split(&quot; + &quot;)
            .map(displayedPhaseTerm)
            .join(&quot;+&quot;);
    }

    function isCollapsedPhase(phase) {
        return collapsedPhaseText(phase) !== phase;
    }

    // Map graph-space (col, row) to SVG pixel coords, with `scale` pixels per
    // unit as [x, y]. col → x, row → y. `topPad` is the vertical offset that
    // accommodates an optional scalar label at the top of the SVG.
    function project(node, minCol, minRow, topPad, scale) {
        return {
            x: LAYOUT.pad + (node.col - minCol) * scale[0],
            y: topPad + (node.row - minRow) * scale[1],
        };
    }

    function renderZx(containerEl, graphJson) {
        if (!containerEl) throw new Error(&quot;renderZx: containerEl is required&quot;);
        if (!graphJson) throw new Error(&quot;renderZx: graphJson is required&quot;);
        const nodeSize = NODE_STYLE.radius;

        // Clear prior content.
        while (containerEl.firstChild) {
            containerEl.removeChild(containerEl.firstChild);
        }

        const nodes = (graphJson.nodes || []).map(function (n) {
            return Object.assign({}, n);
        });
        const rawLinks = graphJson.links || [];
        const scale = graphJson.scale || LAYOUT.scale;

        // Positions arrive already disambiguated (see viz/layout.py), so we
        // render (col, row) as-is — no JS-side staggering needed.

        // Compute (col, row) range.
        let minCol = Infinity, maxCol = -Infinity;
        let minRow = Infinity, maxRow = -Infinity;
        nodes.forEach(function (n) {
            if (n.col &lt; minCol) minCol = n.col;
            if (n.col &gt; maxCol) maxCol = n.col;
            if (n.row &lt; minRow) minRow = n.row;
            if (n.row &gt; maxRow) maxRow = n.row;
        });
        if (!isFinite(minCol)) { minCol = 0; maxCol = 0; minRow = 0; maxRow = 0; }

        // Scalar is rendered as a scrollable HTML div above the SVG, so we
        // no longer reserve vertical space for it inside the SVG itself.
        const showScalar = graphJson.scalar &amp;&amp; graphJson.scalar !== &quot;1&quot; &amp;&amp; graphJson.scalar !== &quot;&quot;;
        const topPad = LAYOUT.pad;

        // Size SVG strictly to content (with small minimum floors so a single
        // node still has visible area).
        const contentW = 2 * LAYOUT.pad + (maxCol - minCol) * scale[0];
        const contentH = topPad + LAYOUT.pad + (maxRow - minRow) * scale[1];
        const w = Math.max(contentW, LAYOUT.minWidth);
        const h = Math.max(contentH, LAYOUT.minHeight);

        // Initial pixel positions.
        const byId = {};
        nodes.forEach(function (n) {
            const p = project(n, minCol, minRow, topPad, scale);
            n.x = p.x;
            n.y = p.y;
            byId[n.id] = n;
        });

        // Resolve link endpoints to node references.
        const links = rawLinks
            .map(function (l) {
                const s = byId[l.source];
                const t = byId[l.target];
                if (!s || !t) return null;
                return { source: s, target: t, t: l.t, domain: l.domain };
            })
            .filter(function (l) { return l !== null; });

        // Scalar rendered as an HTML div above the SVG. Bounding its width to
        // the SVG width and enabling `overflow-x: auto` means long scalars
        // scroll horizontally instead of being clipped by the SVG viewport.
        if (showScalar) {
            const scalarDiv = document.createElement(&quot;div&quot;);
            scalarDiv.textContent = &quot;scalar: &quot; + graphJson.scalar;
            scalarDiv.style.fontFamily = TEXT_STYLE.fontFamily;
            scalarDiv.style.fontSize = TEXT_STYLE.scalar.fontSize;
            scalarDiv.style.color = COLORS.scalarText;
            scalarDiv.style.maxWidth = w + &quot;px&quot;;
            scalarDiv.style.overflowX = TEXT_STYLE.scalar.overflowX;
            scalarDiv.style.whiteSpace = TEXT_STYLE.scalar.whiteSpace;
            scalarDiv.style.padding = TEXT_STYLE.scalar.padding;
            scalarDiv.style.marginBottom = TEXT_STYLE.scalar.marginBottom;
            containerEl.appendChild(scalarDiv);
        }

        const svg = d3.select(containerEl)
            .append(&quot;svg&quot;)
            .attr(&quot;style&quot;, SVG_STYLE)
            .attr(&quot;width&quot;, w)
            .attr(&quot;height&quot;, h);

        // Bend edges that would pass visually through a non-endpoint vertex
        // so it stays unambiguous which vertices an edge actually connects.
        const obstructRadius = nodeSize + NODE_STYLE.obstructionPadding;

        // Spatial index over nodes: each edge queries candidates within its
        // inflated bounding box, cutting obstruction scans from O(N) to O(log N)
        // on average. Kept in sync with node moves by the drag handler.
        const nodeTree = d3.quadtree()
            .x(function (n) { return n.x; })
            .y(function (n) { return n.y; })
            .addAll(nodes);

        // Always emit a quadratic-bezier path (with midpoint control when
        // straight) so d3's default string interpolator can smoothly animate
        // straight ↔ curved transitions during drag.
        function edgePath(edge) {
            const sx = edge.source.x, sy = edge.source.y;
            const tx = edge.target.x, ty = edge.target.y;
            const dx = tx - sx, dy = ty - sy;
            const midX = (sx + tx) / 2, midY = (sy + ty) / 2;
            const L2 = dx * dx + dy * dy;
            if (L2 &lt; EDGE_STYLE.zeroLengthEpsilon) {
                return &quot;M &quot; + sx + &quot;,&quot; + sy + &quot; Q &quot; + midX + &quot;,&quot; + midY + &quot; &quot; + tx + &quot;,&quot; + ty;
            }
            const L = Math.sqrt(L2);

            // Segment AABB inflated by obstructRadius — any obstructing node
            // must fall inside this box.
            const bxMin = Math.min(sx, tx) - obstructRadius;
            const bxMax = Math.max(sx, tx) + obstructRadius;
            const byMin = Math.min(sy, ty) - obstructRadius;
            const byMax = Math.max(sy, ty) + obstructRadius;

            let sumSigned = 0;
            let count = 0;
            nodeTree.visit(function (qnode, qx0, qy0, qx1, qy1) {
                if (qx0 &gt; bxMax || qx1 &lt; bxMin || qy0 &gt; byMax || qy1 &lt; byMin) {
                    return true;  // quadrant can't contain a candidate — prune
                }
                if (!qnode.length) {
                    // Leaf: walk the coincident-points linked list.
                    let leaf = qnode;
                    do {
                        const n = leaf.data;
                        if (n !== edge.source &amp;&amp; n !== edge.target) {
                            const along = ((n.x - sx) * dx + (n.y - sy) * dy) / L2;
                            if (
                                along &gt;= EDGE_STYLE.obstructionRange.min &amp;&amp;
                                along &lt;= EDGE_STYLE.obstructionRange.max
                            ) {
                                const perp = (dx * (n.y - sy) - dy * (n.x - sx)) / L;
                                if (Math.abs(perp) &lt; obstructRadius) {
                                    sumSigned += perp;
                                    count++;
                                }
                            }
                        }
                        leaf = leaf.next;
                    } while (leaf);
                }
                return false;
            });

            let cx = midX, cy = midY;
            if (count &gt; 0) {
                // Offset the control point perpendicular to the segment,
                // away from the side with the heaviest obstruction.
                const sign = sumSigned &gt; 0 ? -1 : 1;
                const offsetMag = obstructRadius + EDGE_STYLE.curveOffset;
                cx = midX + (-dy / L) * sign * offsetMag;
                cy = midY + (dx / L) * sign * offsetMag;
            }
            return &quot;M &quot; + sx + &quot;,&quot; + sy + &quot; Q &quot; + cx + &quot;,&quot; + cy + &quot; &quot; + tx + &quot;,&quot; + ty;
        }

        // Edges. Hadamard edges are drawn as solid blue paths (physics
        // convention); simple edges are solid black.
        const link = svg.append(&quot;g&quot;)
            .attr(&quot;class&quot;, &quot;link&quot;)
            .attr(&quot;fill&quot;, EDGE_STYLE.fill)
            .selectAll(&quot;path&quot;)
            .data(links)
            .enter()
            .append(&quot;path&quot;)
            .attr(&quot;d&quot;, edgePath)
            .attr(&quot;stroke&quot;, function (d) { return edgeColor(d.t); })
            .attr(&quot;stroke-width&quot;, function (d) {
                return isQuantumDomain(d.domain)
                    ? EDGE_STYLE.strokeWidth.quantum
                    : EDGE_STYLE.strokeWidth.classical;
            });

        const node = svg.append(&quot;g&quot;)
            .attr(&quot;class&quot;, &quot;node&quot;)
            .selectAll(&quot;g&quot;)
            .data(nodes)
            .enter()
            .append(&quot;g&quot;)
            .attr(&quot;transform&quot;, function (d) {
                return &quot;translate(&quot; + d.x + &quot;,&quot; + d.y + &quot;)&quot;;
            })
            .attr(&quot;style&quot;, INTERACTION_STYLE.idleCursor);

        // Vertex body.
        node.append(&quot;circle&quot;)
            .attr(&quot;r&quot;, function (d) {
                return nodeRadius(d, nodeSize);
            })
            .attr(&quot;fill&quot;, function (d) { return nodeFill(d.t); })
            .attr(&quot;stroke&quot;, COLORS.outline)
            .attr(&quot;stroke-width&quot;, function (d) {
                return isQuantumDomain(d.domain)
                    ? NODE_STYLE.strokeWidth.quantum
                    : NODE_STYLE.strokeWidth.classical;
            });

        // Label sits just above the node (baseline above the circle's top
        // edge). Non-boundary nodes with a phase will stack the phase above
        // the label; see the phase block below.
        node.filter(function (d) { return d.label; })
            .append(&quot;text&quot;)
            .attr(&quot;y&quot;, function (d) {
                return -nodeRadius(d, nodeSize) - TEXT_STYLE.label.offset;
            })
            .text(function (d) { return d.label; })
            .attr(&quot;text-anchor&quot;, &quot;middle&quot;)
            .attr(&quot;font-size&quot;, TEXT_STYLE.label.fontSize)
            .attr(&quot;font-family&quot;, TEXT_STYLE.fontFamily)
            .attr(&quot;font-weight&quot;, TEXT_STYLE.label.fontWeight)
            .attr(&quot;fill&quot;, COLORS.labelText)
            .attr(&quot;style&quot;, TEXT_STYLE.interaction);

        // Phase label above the node — stacks above the label text when both
        // are present.
        function hasLabel(d) { return !!d.label; }
        function hasPhase(d) { return d.t !== &quot;BOUNDARY&quot; &amp;&amp; d.phase &amp;&amp; d.phase !== &quot;0&quot;; }

        node.filter(hasPhase)
            .append(&quot;text&quot;)
            .attr(&quot;y&quot;, function (d) {
                const radius = nodeRadius(d, nodeSize);
                return hasLabel(d)
                    ? -radius - TEXT_STYLE.phase.stackOffset
                    : -radius - TEXT_STYLE.phase.offset;
            })
            .text(function (d) { return displayedPhaseText(d.phase, false); })
            .attr(&quot;text-anchor&quot;, &quot;middle&quot;)
            .attr(&quot;font-size&quot;, TEXT_STYLE.phase.fontSize)
            .attr(&quot;font-family&quot;, TEXT_STYLE.fontFamily)
            .attr(&quot;fill&quot;, COLORS.phaseText)
            .attr(&quot;style&quot;, function (d) {
                return isCollapsedPhase(d.phase) ? &quot;cursor: default; user-select: none;&quot; : TEXT_STYLE.interaction;
            })
            .on(&quot;mouseenter&quot;, function (event, d) {
                if (isCollapsedPhase(d.phase)) {
                    d3.select(this).text(displayedPhaseText(d.phase, true));
                }
            })
            .on(&quot;mouseleave&quot;, function (event, d) {
                if (isCollapsedPhase(d.phase)) {
                    d3.select(this).text(displayedPhaseText(d.phase, false));
                }
            });

        if (graphJson.show_indices) {
            node.append(&quot;text&quot;)
                .attr(&quot;y&quot;, function (d) {
                    const stacks = (hasPhase(d) ? 1 : 0) + (hasLabel(d) ? 1 : 0);
                    return (
                        -nodeRadius(d, nodeSize) -
                        TEXT_STYLE.index.offset -
                        stacks * TEXT_STYLE.index.stackStep
                    );
                })
                .text(function (d) { return String(d.id); })
                .attr(&quot;text-anchor&quot;, &quot;middle&quot;)
                .attr(&quot;font-size&quot;, TEXT_STYLE.index.fontSize)
                .attr(&quot;font-family&quot;, TEXT_STYLE.fontFamily)
                .attr(&quot;fill&quot;, COLORS.indexText)
                .attr(&quot;style&quot;, TEXT_STYLE.interaction);
        }

        // Drag: update node position and re-render edges. d3 v7 passes
        // (event, datum) to the callback (unlike v5's d3.event).
        //
        // Incident edges are recomputed every tick — their endpoints must
        // track the cursor. Non-incident edges only change because the
        // dragged node may newly obstruct / un-obstruct them, so we batch
        // their refresh through requestAnimationFrame, bounding the cost to
        // one sweep per frame regardless of drag event rate.
        let nonIncidentPending = false;
        let dragTarget = null;

        const drag = d3.drag()
            .on(&quot;start&quot;, function (event, d) {
                d3.select(this).attr(&quot;style&quot;, INTERACTION_STYLE.activeCursor);
                dragTarget = d;
            })
            .on(&quot;drag&quot;, function (event, d) {
                // Keep the spatial index coherent: remove uses the OLD (x, y).
                nodeTree.remove(d);
                d.x += event.dx;
                d.y += event.dy;
                nodeTree.add(d);
                d3.select(this).attr(&quot;transform&quot;, &quot;translate(&quot; + d.x + &quot;,&quot; + d.y + &quot;)&quot;);

                link.filter(function (l) { return l.source === d || l.target === d; })
                    .transition(&quot;edge-follow&quot;)
                    .duration(EDGE_STYLE.followTransitionMs)
                    .ease(d3.easeCubicOut)
                    .attr(&quot;d&quot;, edgePath);

                if (!nonIncidentPending) {
                    nonIncidentPending = true;
                    requestAnimationFrame(function () {
                        nonIncidentPending = false;
                        const target = dragTarget;
                        if (!target) return;
                        link.filter(function (l) { return l.source !== target &amp;&amp; l.target !== target; })
                            .transition(&quot;edge-follow&quot;)
                            .duration(EDGE_STYLE.followTransitionMs)
                            .ease(d3.easeCubicOut)
                            .attr(&quot;d&quot;, edgePath);
                    });
                }
            })
            .on(&quot;end&quot;, function () {
                d3.select(this).attr(&quot;style&quot;, INTERACTION_STYLE.idleCursor);
                dragTarget = null;
            });

        node.call(drag);

        return svg.node();
    }

    // CommonJS-style export for bundlers/tests.
    global.renderZx = renderZx;

(function (global) {
    &quot;use strict&quot;;

    const COLORS = {
        Z: &quot;#ccffcc&quot;,
        X: &quot;#ff8888&quot;,
        BOUNDARY: &quot;#000000&quot;,
        outline: &quot;black&quot;,
        edge: &quot;black&quot;,
        hedge: &quot;#0088ff&quot;,
        scalarText: &quot;#444&quot;,
        labelText: &quot;#000&quot;,
        phaseText: &quot;#00d&quot;,
        indexText: &quot;#999&quot;,
    };

    const LAYOUT = {
        // Pixels per unit of (col, row) when the payload sets none. Matches
        // pyzx's default scale.
        scale: [40, 40],
        // Padding around the graph in pixels.
        pad: 30,
        minWidth: 80,
        minHeight: 60,
    };

    const NODE_STYLE = {
        radius: 8,
        boundaryRadiusScale: {
            classical: 0.5,
            quantum: 0.75,
        },
        strokeWidth: {
            classical: 1.5,
            quantum: 3,
        },
        obstructionPadding: 4,
    };

    const EDGE_STYLE = {
        fill: &quot;none&quot;,
        strokeWidth: {
            classical: 1.5,
            quantum: 3.5,
        },
        obstructionRange: {
            min: 0.1,
            max: 0.9,
        },
        curveOffset: 12,
        zeroLengthEpsilon: 1e-6,
        followTransitionMs: 90,
    };

    const TEXT_STYLE = {
        fontFamily: &quot;monospace&quot;,
        interaction: &quot;pointer-events: none; user-select: none;&quot;,
        label: {
            fontSize: &quot;11px&quot;,
            fontWeight: &quot;600&quot;,
            offset: 4,
        },
        phase: {
            fontSize: &quot;12px&quot;,
            stackOffset: 18,
            offset: 4,
            maxInlineParameters: 3,
        },
        index: {
            fontSize: &quot;10px&quot;,
            offset: 4,
            stackStep: 14,
        },
        scalar: {
            fontSize: &quot;11px&quot;,
            overflowX: &quot;auto&quot;,
            whiteSpace: &quot;nowrap&quot;,
            padding: &quot;2px 0&quot;,
            marginBottom: &quot;4px&quot;,
        },
    };

    const SVG_STYLE = &quot;max-width: none; max-height: none; display: block;&quot;;

    const INTERACTION_STYLE = {
        idleCursor: &quot;cursor: grab;&quot;,
        activeCursor: &quot;cursor: grabbing;&quot;,
    };

    function nodeFill(t) {
        if (t === &quot;Z&quot;) return COLORS.Z;
        if (t === &quot;X&quot;) return COLORS.X;
        return COLORS.BOUNDARY;
    }

    function edgeColor(t) {
        return t === &quot;hadamard&quot; ? COLORS.hedge : COLORS.edge;
    }

    function isQuantumDomain(domain) {
        return domain === &quot;quantum&quot;;
    }

    function nodeRadius(node, nodeSize) {
        if (node.t !== &quot;BOUNDARY&quot;) return nodeSize;
        return isQuantumDomain(node.domain) ? 0.75 * nodeSize : 0.5 * nodeSize;
    }

    function phaseParts(phase) {
        const suffix = &quot;π&quot;;
        if (!phase || !phase.endsWith(suffix)) {
            return null;
        }

        const parenthesizedEnd = &quot;)π&quot;;
        if (phase.endsWith(parenthesizedEnd)) {
            const closeIndex = phase.length - parenthesizedEnd.length;
            const openIndex = phase.lastIndexOf(&quot;(&quot;, closeIndex);
            if (openIndex &gt;= 0) {
                const inner = phase.slice(openIndex + 1, closeIndex);
                const parameters = inner.split(&quot;⊕&quot;).filter(function (p) { return p.length &gt; 0; });
                const concretePrefix = phase.slice(0, openIndex);
                return { concretePrefix: concretePrefix, parameters: parameters };
            }
        }

        const plusIndex = phase.lastIndexOf(&quot; + &quot;);
        if (plusIndex &gt;= 0) {
            return {
                concretePrefix: phase.slice(0, plusIndex + 3),
                parameters: [phase.slice(plusIndex + 3, phase.length - suffix.length)],
            };
        }
        return { concretePrefix: &quot;&quot;, parameters: [phase.slice(0, phase.length - suffix.length)] };
    }

    function collapsedPhaseText(phase) {
        const parts = phaseParts(phase);
        if (!parts || parts.parameters.length &lt;= TEXT_STYLE.phase.maxInlineParameters) {
            return phase;
        }
        const parameters = parts.parameters;
        return (
            parts.concretePrefix +
            parameters[0] +
            &quot;⊕...⊕&quot; +
            parameters[parameters.length - 1] +
            &quot;π&quot;
        );
    }

    // Concrete angles keep their π (&quot;π/4&quot;, &quot;3π/4&quot;, &quot;π&quot;). A parameter always
    // stands for a multiple of π, so parameter terms drop it (&quot;e1π&quot; -&gt; &quot;e1&quot;,
    // &quot;(e1⊕f0)π&quot; -&gt; &quot;e1⊕f0&quot;).
    function displayedPhaseTerm(term) {
        if (!term || !term.endsWith(&quot;π&quot;)) {
            return term;
        }
        const withoutPi = term.slice(0, -1);
        if (/^~?\d*$/.test(withoutPi)) {
            return term;
        }
        if (withoutPi.startsWith(&quot;(&quot;) &amp;&amp; withoutPi.endsWith(&quot;)&quot;)) {
            return withoutPi.slice(1, -1);
        }
        return withoutPi;
    }

    function displayedPhaseText(phase, expanded) {
        return (expanded ? phase : collapsedPhaseText(phase))
            .split(&quot; + &quot;)
            .map(displayedPhaseTerm)
            .join(&quot;+&quot;);
    }

    function isCollapsedPhase(phase) {
        return collapsedPhaseText(phase) !== phase;
    }

    // Map graph-space (col, row) to SVG pixel coords, with `scale` pixels per
    // unit as [x, y]. col → x, row → y. `topPad` is the vertical offset that
    // accommodates an optional scalar label at the top of the SVG.
    function project(node, minCol, minRow, topPad, scale) {
        return {
            x: LAYOUT.pad + (node.col - minCol) * scale[0],
            y: topPad + (node.row - minRow) * scale[1],
        };
    }

    function renderZx(containerEl, graphJson) {
        if (!containerEl) throw new Error(&quot;renderZx: containerEl is required&quot;);
        if (!graphJson) throw new Error(&quot;renderZx: graphJson is required&quot;);
        const nodeSize = NODE_STYLE.radius;

        // Clear prior content.
        while (containerEl.firstChild) {
            containerEl.removeChild(containerEl.firstChild);
        }

        const nodes = (graphJson.nodes || []).map(function (n) {
            return Object.assign({}, n);
        });
        const rawLinks = graphJson.links || [];
        const scale = graphJson.scale || LAYOUT.scale;

        // Positions arrive already disambiguated (see viz/layout.py), so we
        // render (col, row) as-is — no JS-side staggering needed.

        // Compute (col, row) range.
        let minCol = Infinity, maxCol = -Infinity;
        let minRow = Infinity, maxRow = -Infinity;
        nodes.forEach(function (n) {
            if (n.col &lt; minCol) minCol = n.col;
            if (n.col &gt; maxCol) maxCol = n.col;
            if (n.row &lt; minRow) minRow = n.row;
            if (n.row &gt; maxRow) maxRow = n.row;
        });
        if (!isFinite(minCol)) { minCol = 0; maxCol = 0; minRow = 0; maxRow = 0; }

        // Scalar is rendered as a scrollable HTML div above the SVG, so we
        // no longer reserve vertical space for it inside the SVG itself.
        const showScalar = graphJson.scalar &amp;&amp; graphJson.scalar !== &quot;1&quot; &amp;&amp; graphJson.scalar !== &quot;&quot;;
        const topPad = LAYOUT.pad;

        // Size SVG strictly to content (with small minimum floors so a single
        // node still has visible area).
        const contentW = 2 * LAYOUT.pad + (maxCol - minCol) * scale[0];
        const contentH = topPad + LAYOUT.pad + (maxRow - minRow) * scale[1];
        const w = Math.max(contentW, LAYOUT.minWidth);
        const h = Math.max(contentH, LAYOUT.minHeight);

        // Initial pixel positions.
        const byId = {};
        nodes.forEach(function (n) {
            const p = project(n, minCol, minRow, topPad, scale);
            n.x = p.x;
            n.y = p.y;
            byId[n.id] = n;
        });

        // Resolve link endpoints to node references.
        const links = rawLinks
            .map(function (l) {
                const s = byId[l.source];
                const t = byId[l.target];
                if (!s || !t) return null;
                return { source: s, target: t, t: l.t, domain: l.domain };
            })
            .filter(function (l) { return l !== null; });

        // Scalar rendered as an HTML div above the SVG. Bounding its width to
        // the SVG width and enabling `overflow-x: auto` means long scalars
        // scroll horizontally instead of being clipped by the SVG viewport.
        if (showScalar) {
            const scalarDiv = document.createElement(&quot;div&quot;);
            scalarDiv.textContent = &quot;scalar: &quot; + graphJson.scalar;
            scalarDiv.style.fontFamily = TEXT_STYLE.fontFamily;
            scalarDiv.style.fontSize = TEXT_STYLE.scalar.fontSize;
            scalarDiv.style.color = COLORS.scalarText;
            scalarDiv.style.maxWidth = w + &quot;px&quot;;
            scalarDiv.style.overflowX = TEXT_STYLE.scalar.overflowX;
            scalarDiv.style.whiteSpace = TEXT_STYLE.scalar.whiteSpace;
            scalarDiv.style.padding = TEXT_STYLE.scalar.padding;
            scalarDiv.style.marginBottom = TEXT_STYLE.scalar.marginBottom;
            containerEl.appendChild(scalarDiv);
        }

        const svg = d3.select(containerEl)
            .append(&quot;svg&quot;)
            .attr(&quot;style&quot;, SVG_STYLE)
            .attr(&quot;width&quot;, w)
            .attr(&quot;height&quot;, h);

        // Bend edges that would pass visually through a non-endpoint vertex
        // so it stays unambiguous which vertices an edge actually connects.
        const obstructRadius = nodeSize + NODE_STYLE.obstructionPadding;

        // Spatial index over nodes: each edge queries candidates within its
        // inflated bounding box, cutting obstruction scans from O(N) to O(log N)
        // on average. Kept in sync with node moves by the drag handler.
        const nodeTree = d3.quadtree()
            .x(function (n) { return n.x; })
            .y(function (n) { return n.y; })
            .addAll(nodes);

        // Always emit a quadratic-bezier path (with midpoint control when
        // straight) so d3's default string interpolator can smoothly animate
        // straight ↔ curved transitions during drag.
        function edgePath(edge) {
            const sx = edge.source.x, sy = edge.source.y;
            const tx = edge.target.x, ty = edge.target.y;
            const dx = tx - sx, dy = ty - sy;
            const midX = (sx + tx) / 2, midY = (sy + ty) / 2;
            const L2 = dx * dx + dy * dy;
            if (L2 &lt; EDGE_STYLE.zeroLengthEpsilon) {
                return &quot;M &quot; + sx + &quot;,&quot; + sy + &quot; Q &quot; + midX + &quot;,&quot; + midY + &quot; &quot; + tx + &quot;,&quot; + ty;
            }
            const L = Math.sqrt(L2);

            // Segment AABB inflated by obstructRadius — any obstructing node
            // must fall inside this box.
            const bxMin = Math.min(sx, tx) - obstructRadius;
            const bxMax = Math.max(sx, tx) + obstructRadius;
            const byMin = Math.min(sy, ty) - obstructRadius;
            const byMax = Math.max(sy, ty) + obstructRadius;

            let sumSigned = 0;
            let count = 0;
            nodeTree.visit(function (qnode, qx0, qy0, qx1, qy1) {
                if (qx0 &gt; bxMax || qx1 &lt; bxMin || qy0 &gt; byMax || qy1 &lt; byMin) {
                    return true;  // quadrant can't contain a candidate — prune
                }
                if (!qnode.length) {
                    // Leaf: walk the coincident-points linked list.
                    let leaf = qnode;
                    do {
                        const n = leaf.data;
                        if (n !== edge.source &amp;&amp; n !== edge.target) {
                            const along = ((n.x - sx) * dx + (n.y - sy) * dy) / L2;
                            if (
                                along &gt;= EDGE_STYLE.obstructionRange.min &amp;&amp;
                                along &lt;= EDGE_STYLE.obstructionRange.max
                            ) {
                                const perp = (dx * (n.y - sy) - dy * (n.x - sx)) / L;
                                if (Math.abs(perp) &lt; obstructRadius) {
                                    sumSigned += perp;
                                    count++;
                                }
                            }
                        }
                        leaf = leaf.next;
                    } while (leaf);
                }
                return false;
            });

            let cx = midX, cy = midY;
            if (count &gt; 0) {
                // Offset the control point perpendicular to the segment,
                // away from the side with the heaviest obstruction.
                const sign = sumSigned &gt; 0 ? -1 : 1;
                const offsetMag = obstructRadius + EDGE_STYLE.curveOffset;
                cx = midX + (-dy / L) * sign * offsetMag;
                cy = midY + (dx / L) * sign * offsetMag;
            }
            return &quot;M &quot; + sx + &quot;,&quot; + sy + &quot; Q &quot; + cx + &quot;,&quot; + cy + &quot; &quot; + tx + &quot;,&quot; + ty;
        }

        // Edges. Hadamard edges are drawn as solid blue paths (physics
        // convention); simple edges are solid black.
        const link = svg.append(&quot;g&quot;)
            .attr(&quot;class&quot;, &quot;link&quot;)
            .attr(&quot;fill&quot;, EDGE_STYLE.fill)
            .selectAll(&quot;path&quot;)
            .data(links)
            .enter()
            .append(&quot;path&quot;)
            .attr(&quot;d&quot;, edgePath)
            .attr(&quot;stroke&quot;, function (d) { return edgeColor(d.t); })
            .attr(&quot;stroke-width&quot;, function (d) {
                return isQuantumDomain(d.domain)
                    ? EDGE_STYLE.strokeWidth.quantum
                    : EDGE_STYLE.strokeWidth.classical;
            });

        const node = svg.append(&quot;g&quot;)
            .attr(&quot;class&quot;, &quot;node&quot;)
            .selectAll(&quot;g&quot;)
            .data(nodes)
            .enter()
            .append(&quot;g&quot;)
            .attr(&quot;transform&quot;, function (d) {
                return &quot;translate(&quot; + d.x + &quot;,&quot; + d.y + &quot;)&quot;;
            })
            .attr(&quot;style&quot;, INTERACTION_STYLE.idleCursor);

        // Vertex body.
        node.append(&quot;circle&quot;)
            .attr(&quot;r&quot;, function (d) {
                return nodeRadius(d, nodeSize);
            })
            .attr(&quot;fill&quot;, function (d) { return nodeFill(d.t); })
            .attr(&quot;stroke&quot;, COLORS.outline)
            .attr(&quot;stroke-width&quot;, function (d) {
                return isQuantumDomain(d.domain)
                    ? NODE_STYLE.strokeWidth.quantum
                    : NODE_STYLE.strokeWidth.classical;
            });

        // Label sits just above the node (baseline above the circle's top
        // edge). Non-boundary nodes with a phase will stack the phase above
        // the label; see the phase block below.
        node.filter(function (d) { return d.label; })
            .append(&quot;text&quot;)
            .attr(&quot;y&quot;, function (d) {
                return -nodeRadius(d, nodeSize) - TEXT_STYLE.label.offset;
            })
            .text(function (d) { return d.label; })
            .attr(&quot;text-anchor&quot;, &quot;middle&quot;)
            .attr(&quot;font-size&quot;, TEXT_STYLE.label.fontSize)
            .attr(&quot;font-family&quot;, TEXT_STYLE.fontFamily)
            .attr(&quot;font-weight&quot;, TEXT_STYLE.label.fontWeight)
            .attr(&quot;fill&quot;, COLORS.labelText)
            .attr(&quot;style&quot;, TEXT_STYLE.interaction);

        // Phase label above the node — stacks above the label text when both
        // are present.
        function hasLabel(d) { return !!d.label; }
        function hasPhase(d) { return d.t !== &quot;BOUNDARY&quot; &amp;&amp; d.phase &amp;&amp; d.phase !== &quot;0&quot;; }

        node.filter(hasPhase)
            .append(&quot;text&quot;)
            .attr(&quot;y&quot;, function (d) {
                const radius = nodeRadius(d, nodeSize);
                return hasLabel(d)
                    ? -radius - TEXT_STYLE.phase.stackOffset
                    : -radius - TEXT_STYLE.phase.offset;
            })
            .text(function (d) { return displayedPhaseText(d.phase, false); })
            .attr(&quot;text-anchor&quot;, &quot;middle&quot;)
            .attr(&quot;font-size&quot;, TEXT_STYLE.phase.fontSize)
            .attr(&quot;font-family&quot;, TEXT_STYLE.fontFamily)
            .attr(&quot;fill&quot;, COLORS.phaseText)
            .attr(&quot;style&quot;, function (d) {
                return isCollapsedPhase(d.phase) ? &quot;cursor: default; user-select: none;&quot; : TEXT_STYLE.interaction;
            })
            .on(&quot;mouseenter&quot;, function (event, d) {
                if (isCollapsedPhase(d.phase)) {
                    d3.select(this).text(displayedPhaseText(d.phase, true));
                }
            })
            .on(&quot;mouseleave&quot;, function (event, d) {
                if (isCollapsedPhase(d.phase)) {
                    d3.select(this).text(displayedPhaseText(d.phase, false));
                }
            });

        if (graphJson.show_indices) {
            node.append(&quot;text&quot;)
                .attr(&quot;y&quot;, function (d) {
                    const stacks = (hasPhase(d) ? 1 : 0) + (hasLabel(d) ? 1 : 0);
                    return (
                        -nodeRadius(d, nodeSize) -
                        TEXT_STYLE.index.offset -
                        stacks * TEXT_STYLE.index.stackStep
                    );
                })
                .text(function (d) { return String(d.id); })
                .attr(&quot;text-anchor&quot;, &quot;middle&quot;)
                .attr(&quot;font-size&quot;, TEXT_STYLE.index.fontSize)
                .attr(&quot;font-family&quot;, TEXT_STYLE.fontFamily)
                .attr(&quot;fill&quot;, COLORS.indexText)
                .attr(&quot;style&quot;, TEXT_STYLE.interaction);
        }

        // Drag: update node position and re-render edges. d3 v7 passes
        // (event, datum) to the callback (unlike v5's d3.event).
        //
        // Incident edges are recomputed every tick — their endpoints must
        // track the cursor. Non-incident edges only change because the
        // dragged node may newly obstruct / un-obstruct them, so we batch
        // their refresh through requestAnimationFrame, bounding the cost to
        // one sweep per frame regardless of drag event rate.
        let nonIncidentPending = false;
        let dragTarget = null;

        const drag = d3.drag()
            .on(&quot;start&quot;, function (event, d) {
                d3.select(this).attr(&quot;style&quot;, INTERACTION_STYLE.activeCursor);
                dragTarget = d;
            })
            .on(&quot;drag&quot;, function (event, d) {
                // Keep the spatial index coherent: remove uses the OLD (x, y).
                nodeTree.remove(d);
                d.x += event.dx;
                d.y += event.dy;
                nodeTree.add(d);
                d3.select(this).attr(&quot;transform&quot;, &quot;translate(&quot; + d.x + &quot;,&quot; + d.y + &quot;)&quot;);

                link.filter(function (l) { return l.source === d || l.target === d; })
                    .transition(&quot;edge-follow&quot;)
                    .duration(EDGE_STYLE.followTransitionMs)
                    .ease(d3.easeCubicOut)
                    .attr(&quot;d&quot;, edgePath);

                if (!nonIncidentPending) {
                    nonIncidentPending = true;
                    requestAnimationFrame(function () {
                        nonIncidentPending = false;
                        const target = dragTarget;
                        if (!target) return;
                        link.filter(function (l) { return l.source !== target &amp;&amp; l.target !== target; })
                            .transition(&quot;edge-follow&quot;)
                            .duration(EDGE_STYLE.followTransitionMs)
                            .ease(d3.easeCubicOut)
                            .attr(&quot;d&quot;, edgePath);
                    });
                }
            })
            .on(&quot;end&quot;, function () {
                d3.select(this).attr(&quot;style&quot;, INTERACTION_STYLE.idleCursor);
                dragTarget = null;
            });

        node.call(drag);

        return svg.node();
    }

    // Expose renderZx both globally (for inline &lt;script&gt; usage) and as a
    // CommonJS-style export for bundlers/tests.
    global.renderZx = renderZx;
    if (typeof module !== &quot;undefined&quot; &amp;&amp; module.exports) {
        module.exports = { renderZx: renderZx, displayedPhaseText: displayedPhaseText };
    }

window.d3 = d3;
// zxparam D3 viewer

//   - Vertex types: Z, X, BOUNDARY

(function (global) {
    &quot;use strict&quot;;

    const COLORS = {
        Z: &quot;#ccffcc&quot;,
        X: &quot;#ff8888&quot;,
        BOUNDARY: &quot;#000000&quot;,
        outline: &quot;black&quot;,
        edge: &quot;black&quot;,
        hedge: &quot;#0088ff&quot;,
        scalarText: &quot;#444&quot;,
        labelText: &quot;#000&quot;,
        phaseText: &quot;#00d&quot;,
        indexText: &quot;#999&quot;,
    };

    const LAYOUT = {
        // Pixels per unit of (col, row) when the payload sets none. Matches
        // pyzx's default scale.
        scale: [40, 40],
        // Padding around the graph in pixels.
        pad: 30,
        minWidth: 80,
        minHeight: 60,
    };

    const NODE_STYLE = {
        radius: 8,
        boundaryRadiusScale: {
            classical: 0.5,
            quantum: 0.75,
        },
        strokeWidth: {
            classical: 1.5,
            quantum: 3,
        },
        obstructionPadding: 4,
    };

    const EDGE_STYLE = {
        fill: &quot;none&quot;,
        strokeWidth: {
            classical: 1.5,
            quantum: 3.5,
        },
        obstructionRange: {
            min: 0.1,
            max: 0.9,
        },
        curveOffset: 12,
        zeroLengthEpsilon: 1e-6,
        followTransitionMs: 90,
    };

    const TEXT_STYLE = {
        fontFamily: &quot;monospace&quot;,
        interaction: &quot;pointer-events: none; user-select: none;&quot;,
        label: {
            fontSize: &quot;11px&quot;,
            fontWeight: &quot;600&quot;,
            offset: 4,
        },
        phase: {
            fontSize: &quot;12px&quot;,
            stackOffset: 18,
            offset: 4,
            maxInlineParameters: 3,
        },
        index: {
            fontSize: &quot;10px&quot;,
            offset: 4,
            stackStep: 14,
        },
        scalar: {
            fontSize: &quot;11px&quot;,
            overflowX: &quot;auto&quot;,
            whiteSpace: &quot;nowrap&quot;,
            padding: &quot;2px 0&quot;,
            marginBottom: &quot;4px&quot;,
        },
    };

    const SVG_STYLE = &quot;max-width: none; max-height: none; display: block;&quot;;

    const INTERACTION_STYLE = {
        idleCursor: &quot;cursor: grab;&quot;,
        activeCursor: &quot;cursor: grabbing;&quot;,
    };

    function nodeFill(t) {
        if (t === &quot;Z&quot;) return COLORS.Z;
        if (t === &quot;X&quot;) return COLORS.X;
        return COLORS.BOUNDARY;
    }

    function edgeColor(t) {
        return t === &quot;hadamard&quot; ? COLORS.hedge : COLORS.edge;
    }

    function isQuantumDomain(domain) {
        return domain === &quot;quantum&quot;;
    }

    function nodeRadius(node, nodeSize) {
        if (node.t !== &quot;BOUNDARY&quot;) return nodeSize;
        return isQuantumDomain(node.domain) ? 0.75 * nodeSize : 0.5 * nodeSize;
    }

    function phaseParts(phase) {
        const suffix = &quot;π&quot;;
        if (!phase || !phase.endsWith(suffix)) {
            return null;
        }

        const parenthesizedEnd = &quot;)π&quot;;
        if (phase.endsWith(parenthesizedEnd)) {
            const closeIndex = phase.length - parenthesizedEnd.length;
            const openIndex = phase.lastIndexOf(&quot;(&quot;, closeIndex);
            if (openIndex &gt;= 0) {
                const inner = phase.slice(openIndex + 1, closeIndex);
                const parameters = inner.split(&quot;⊕&quot;).filter(function (p) { return p.length &gt; 0; });
                const concretePrefix = phase.slice(0, openIndex);
                return { concretePrefix: concretePrefix, parameters: parameters };
            }
        }

        const plusIndex = phase.lastIndexOf(&quot; + &quot;);
        if (plusIndex &gt;= 0) {
            return {
                concretePrefix: phase.slice(0, plusIndex + 3),
                parameters: [phase.slice(plusIndex + 3, phase.length - suffix.length)],
            };
        }
        return { concretePrefix: &quot;&quot;, parameters: [phase.slice(0, phase.length - suffix.length)] };
    }

    function collapsedPhaseText(phase) {
        const parts = phaseParts(phase);
        if (!parts || parts.parameters.length &lt;= TEXT_STYLE.phase.maxInlineParameters) {
            return phase;
        }
        const parameters = parts.parameters;
        return (
            parts.concretePrefix +
            parameters[0] +
            &quot;⊕...⊕&quot; +
            parameters[parameters.length - 1] +
            &quot;π&quot;
        );
    }

    // Concrete angles keep their π (&quot;π/4&quot;, &quot;3π/4&quot;, &quot;π&quot;). A parameter always
    // stands for a multiple of π, so parameter terms drop it (&quot;e1π&quot; -&gt; &quot;e1&quot;,
    // &quot;(e1⊕f0)π&quot; -&gt; &quot;e1⊕f0&quot;).
    function displayedPhaseTerm(term) {
        if (!term || !term.endsWith(&quot;π&quot;)) {
            return term;
        }
        const withoutPi = term.slice(0, -1);
        if (/^~?\d*$/.test(withoutPi)) {
            return term;
        }
        if (withoutPi.startsWith(&quot;(&quot;) &amp;&amp; withoutPi.endsWith(&quot;)&quot;)) {
            return withoutPi.slice(1, -1);
        }
        return withoutPi;
    }

    function displayedPhaseText(phase, expanded) {
        return (expanded ? phase : collapsedPhaseText(phase))
            .split(&quot; + &quot;)
            .map(displayedPhaseTerm)
            .join(&quot;+&quot;);
    }

    function isCollapsedPhase(phase) {
        return collapsedPhaseText(phase) !== phase;
    }

    // Map graph-space (col, row) to SVG pixel coords, with `scale` pixels per
    // unit as [x, y]. col → x, row → y. `topPad` is the vertical offset that
    // accommodates an optional scalar label at the top of the SVG.
    function project(node, minCol, minRow, topPad, scale) {
        return {
            x: LAYOUT.pad + (node.col - minCol) * scale[0],
            y: topPad + (node.row - minRow) * scale[1],
        };
    }

    function renderZx(containerEl, graphJson) {
        if (!containerEl) throw new Error(&quot;renderZx: containerEl is required&quot;);
        if (!graphJson) throw new Error(&quot;renderZx: graphJson is required&quot;);
        const nodeSize = NODE_STYLE.radius;

        // Clear prior content.
        while (containerEl.firstChild) {
            containerEl.removeChild(containerEl.firstChild);
        }

        const nodes = (graphJson.nodes || []).map(function (n) {
            return Object.assign({}, n);
        });
        const rawLinks = graphJson.links || [];
        const scale = graphJson.scale || LAYOUT.scale;

        // Positions arrive already disambiguated (see viz/layout.py), so we
        // render (col, row) as-is — no JS-side staggering needed.

        // Compute (col, row) range.
        let minCol = Infinity, maxCol = -Infinity;
        let minRow = Infinity, maxRow = -Infinity;
        nodes.forEach(function (n) {
            if (n.col &lt; minCol) minCol = n.col;
            if (n.col &gt; maxCol) maxCol = n.col;
            if (n.row &lt; minRow) minRow = n.row;
            if (n.row &gt; maxRow) maxRow = n.row;
        });
        if (!isFinite(minCol)) { minCol = 0; maxCol = 0; minRow = 0; maxRow = 0; }

        // Scalar is rendered as a scrollable HTML div above the SVG, so we
        // no longer reserve vertical space for it inside the SVG itself.
        const showScalar = graphJson.scalar &amp;&amp; graphJson.scalar !== &quot;1&quot; &amp;&amp; graphJson.scalar !== &quot;&quot;;
        const topPad = LAYOUT.pad;

        // Size SVG strictly to content (with small minimum floors so a single
        // node still has visible area).
        const contentW = 2 * LAYOUT.pad + (maxCol - minCol) * scale[0];
        const contentH = topPad + LAYOUT.pad + (maxRow - minRow) * scale[1];
        const w = Math.max(contentW, LAYOUT.minWidth);
        const h = Math.max(contentH, LAYOUT.minHeight);

        // Initial pixel positions.
        const byId = {};
        nodes.forEach(function (n) {
            const p = project(n, minCol, minRow, topPad, scale);
            n.x = p.x;
            n.y = p.y;
            byId[n.id] = n;
        });

        // Resolve link endpoints to node references.
        const links = rawLinks
            .map(function (l) {
                const s = byId[l.source];
                const t = byId[l.target];
                if (!s || !t) return null;
                return { source: s, target: t, t: l.t, domain: l.domain };
            })
            .filter(function (l) { return l !== null; });

        // Scalar rendered as an HTML div above the SVG. Bounding its width to
        // the SVG width and enabling `overflow-x: auto` means long scalars
        // scroll horizontally instead of being clipped by the SVG viewport.
        if (showScalar) {
            const scalarDiv = document.createElement(&quot;div&quot;);
            scalarDiv.textContent = &quot;scalar: &quot; + graphJson.scalar;
            scalarDiv.style.fontFamily = TEXT_STYLE.fontFamily;
            scalarDiv.style.fontSize = TEXT_STYLE.scalar.fontSize;
            scalarDiv.style.color = COLORS.scalarText;
            scalarDiv.style.maxWidth = w + &quot;px&quot;;
            scalarDiv.style.overflowX = TEXT_STYLE.scalar.overflowX;
            scalarDiv.style.whiteSpace = TEXT_STYLE.scalar.whiteSpace;
            scalarDiv.style.padding = TEXT_STYLE.scalar.padding;
            scalarDiv.style.marginBottom = TEXT_STYLE.scalar.marginBottom;
            containerEl.appendChild(scalarDiv);
        }

        const svg = d3.select(containerEl)
            .append(&quot;svg&quot;)
            .attr(&quot;style&quot;, SVG_STYLE)
            .attr(&quot;width&quot;, w)
            .attr(&quot;height&quot;, h);

        // Bend edges that would pass visually through a non-endpoint vertex
        // so it stays unambiguous which vertices an edge actually connects.
        const obstructRadius = nodeSize + NODE_STYLE.obstructionPadding;

        // Spatial index over nodes: each edge queries candidates within its
        // inflated bounding box, cutting obstruction scans from O(N) to O(log N)
        // on average. Kept in sync with node moves by the drag handler.
        const nodeTree = d3.quadtree()
            .x(function (n) { return n.x; })
            .y(function (n) { return n.y; })
            .addAll(nodes);

        // Always emit a quadratic-bezier path (with midpoint control when
        // straight) so d3's default string interpolator can smoothly animate
        // straight ↔ curved transitions during drag.
        function edgePath(edge) {
            const sx = edge.source.x, sy = edge.source.y;
            const tx = edge.target.x, ty = edge.target.y;
            const dx = tx - sx, dy = ty - sy;
            const midX = (sx + tx) / 2, midY = (sy + ty) / 2;
            const L2 = dx * dx + dy * dy;
            if (L2 &lt; EDGE_STYLE.zeroLengthEpsilon) {
                return &quot;M &quot; + sx + &quot;,&quot; + sy + &quot; Q &quot; + midX + &quot;,&quot; + midY + &quot; &quot; + tx + &quot;,&quot; + ty;
            }
            const L = Math.sqrt(L2);

            // Segment AABB inflated by obstructRadius — any obstructing node
            // must fall inside this box.
            const bxMin = Math.min(sx, tx) - obstructRadius;
            const bxMax = Math.max(sx, tx) + obstructRadius;
            const byMin = Math.min(sy, ty) - obstructRadius;
            const byMax = Math.max(sy, ty) + obstructRadius;

            let sumSigned = 0;
            let count = 0;
            nodeTree.visit(function (qnode, qx0, qy0, qx1, qy1) {
                if (qx0 &gt; bxMax || qx1 &lt; bxMin || qy0 &gt; byMax || qy1 &lt; byMin) {
                    return true;  // quadrant can't contain a candidate — prune
                }
                if (!qnode.length) {
                    // Leaf: walk the coincident-points linked list.
                    let leaf = qnode;
                    do {
                        const n = leaf.data;
                        if (n !== edge.source &amp;&amp; n !== edge.target) {
                            const along = ((n.x - sx) * dx + (n.y - sy) * dy) / L2;
                            if (
                                along &gt;= EDGE_STYLE.obstructionRange.min &amp;&amp;
                                along &lt;= EDGE_STYLE.obstructionRange.max
                            ) {
                                const perp = (dx * (n.y - sy) - dy * (n.x - sx)) / L;
                                if (Math.abs(perp) &lt; obstructRadius) {
                                    sumSigned += perp;
                                    count++;
                                }
                            }
                        }
                        leaf = leaf.next;
                    } while (leaf);
                }
                return false;
            });

            let cx = midX, cy = midY;
            if (count &gt; 0) {
                // Offset the control point perpendicular to the segment,
                // away from the side with the heaviest obstruction.
                const sign = sumSigned &gt; 0 ? -1 : 1;
                const offsetMag = obstructRadius + EDGE_STYLE.curveOffset;
                cx = midX + (-dy / L) * sign * offsetMag;
                cy = midY + (dx / L) * sign * offsetMag;
            }
            return &quot;M &quot; + sx + &quot;,&quot; + sy + &quot; Q &quot; + cx + &quot;,&quot; + cy + &quot; &quot; + tx + &quot;,&quot; + ty;
        }

        // Edges. Hadamard edges are drawn as solid blue paths (physics
        // convention); simple edges are solid black.
        const link = svg.append(&quot;g&quot;)
            .attr(&quot;class&quot;, &quot;link&quot;)
            .attr(&quot;fill&quot;, EDGE_STYLE.fill)
            .selectAll(&quot;path&quot;)
            .data(links)
            .enter()
            .append(&quot;path&quot;)
            .attr(&quot;d&quot;, edgePath)
            .attr(&quot;stroke&quot;, function (d) { return edgeColor(d.t); })
            .attr(&quot;stroke-width&quot;, function (d) {
                return isQuantumDomain(d.domain)
                    ? EDGE_STYLE.strokeWidth.quantum
                    : EDGE_STYLE.strokeWidth.classical;
            });

        const node = svg.append(&quot;g&quot;)
            .attr(&quot;class&quot;, &quot;node&quot;)
            .selectAll(&quot;g&quot;)
            .data(nodes)
            .enter()
            .append(&quot;g&quot;)
            .attr(&quot;transform&quot;, function (d) {
                return &quot;translate(&quot; + d.x + &quot;,&quot; + d.y + &quot;)&quot;;
            })
            .attr(&quot;style&quot;, INTERACTION_STYLE.idleCursor);

        // Vertex body.
        node.append(&quot;circle&quot;)
            .attr(&quot;r&quot;, function (d) {
                return nodeRadius(d, nodeSize);
            })
            .attr(&quot;fill&quot;, function (d) { return nodeFill(d.t); })
            .attr(&quot;stroke&quot;, COLORS.outline)
            .attr(&quot;stroke-width&quot;, function (d) {
                return isQuantumDomain(d.domain)
                    ? NODE_STYLE.strokeWidth.quantum
                    : NODE_STYLE.strokeWidth.classical;
            });

        // Label sits just above the node (baseline above the circle's top
        // edge). Non-boundary nodes with a phase will stack the phase above
        // the label; see the phase block below.
        node.filter(function (d) { return d.label; })
            .append(&quot;text&quot;)
            .attr(&quot;y&quot;, function (d) {
                return -nodeRadius(d, nodeSize) - TEXT_STYLE.label.offset;
            })
            .text(function (d) { return d.label; })
            .attr(&quot;text-anchor&quot;, &quot;middle&quot;)
            .attr(&quot;font-size&quot;, TEXT_STYLE.label.fontSize)
            .attr(&quot;font-family&quot;, TEXT_STYLE.fontFamily)
            .attr(&quot;font-weight&quot;, TEXT_STYLE.label.fontWeight)
            .attr(&quot;fill&quot;, COLORS.labelText)
            .attr(&quot;style&quot;, TEXT_STYLE.interaction);

        // Phase label above the node — stacks above the label text when both
        // are present.
        function hasLabel(d) { return !!d.label; }
        function hasPhase(d) { return d.t !== &quot;BOUNDARY&quot; &amp;&amp; d.phase &amp;&amp; d.phase !== &quot;0&quot;; }

        node.filter(hasPhase)
            .append(&quot;text&quot;)
            .attr(&quot;y&quot;, function (d) {
                const radius = nodeRadius(d, nodeSize);
                return hasLabel(d)
                    ? -radius - TEXT_STYLE.phase.stackOffset
                    : -radius - TEXT_STYLE.phase.offset;
            })
            .text(function (d) { return displayedPhaseText(d.phase, false); })
            .attr(&quot;text-anchor&quot;, &quot;middle&quot;)
            .attr(&quot;font-size&quot;, TEXT_STYLE.phase.fontSize)
            .attr(&quot;font-family&quot;, TEXT_STYLE.fontFamily)
            .attr(&quot;fill&quot;, COLORS.phaseText)
            .attr(&quot;style&quot;, function (d) {
                return isCollapsedPhase(d.phase) ? &quot;cursor: default; user-select: none;&quot; : TEXT_STYLE.interaction;
            })
            .on(&quot;mouseenter&quot;, function (event, d) {
                if (isCollapsedPhase(d.phase)) {
                    d3.select(this).text(displayedPhaseText(d.phase, true));
                }
            })
            .on(&quot;mouseleave&quot;, function (event, d) {
                if (isCollapsedPhase(d.phase)) {
                    d3.select(this).text(displayedPhaseText(d.phase, false));
                }
            });

        if (graphJson.show_indices) {
            node.append(&quot;text&quot;)
                .attr(&quot;y&quot;, function (d) {
                    const stacks = (hasPhase(d) ? 1 : 0) + (hasLabel(d) ? 1 : 0);
                    return (
                        -nodeRadius(d, nodeSize) -
                        TEXT_STYLE.index.offset -
                        stacks * TEXT_STYLE.index.stackStep
                    );
                })
                .text(function (d) { return String(d.id); })
                .attr(&quot;text-anchor&quot;, &quot;middle&quot;)
                .attr(&quot;font-size&quot;, TEXT_STYLE.index.fontSize)
                .attr(&quot;font-family&quot;, TEXT_STYLE.fontFamily)
                .attr(&quot;fill&quot;, COLORS.indexText)
                .attr(&quot;style&quot;, TEXT_STYLE.interaction);
        }

        // Drag: update node position and re-render edges. d3 v7 passes
        // (event, datum) to the callback (unlike v5's d3.event).
        //
        // Incident edges are recomputed every tick — their endpoints must
        // track the cursor. Non-incident edges only change because the
        // dragged node may newly obstruct / un-obstruct them, so we batch
        // their refresh through requestAnimationFrame, bounding the cost to
        // one sweep per frame regardless of drag event rate.
        let nonIncidentPending = false;
        let dragTarget = null;

        const drag = d3.drag()
            .on(&quot;start&quot;, function (event, d) {
                d3.select(this).attr(&quot;style&quot;, INTERACTION_STYLE.activeCursor);
                dragTarget = d;
            })
            .on(&quot;drag&quot;, function (event, d) {
                // Keep the spatial index coherent: remove uses the OLD (x, y).
                nodeTree.remove(d);
                d.x += event.dx;
                d.y += event.dy;
                nodeTree.add(d);
                d3.select(this).attr(&quot;transform&quot;, &quot;translate(&quot; + d.x + &quot;,&quot; + d.y + &quot;)&quot;);

                link.filter(function (l) { return l.source === d || l.target === d; })
                    .transition(&quot;edge-follow&quot;)
                    .duration(EDGE_STYLE.followTransitionMs)
                    .ease(d3.easeCubicOut)
                    .attr(&quot;d&quot;, edgePath);

                if (!nonIncidentPending) {
                    nonIncidentPending = true;
                    requestAnimationFrame(function () {
                        nonIncidentPending = false;
                        const target = dragTarget;
                        if (!target) return;
                        link.filter(function (l) { return l.source !== target &amp;&amp; l.target !== target; })
                            .transition(&quot;edge-follow&quot;)
                            .duration(EDGE_STYLE.followTransitionMs)
                            .ease(d3.easeCubicOut)
                            .attr(&quot;d&quot;, edgePath);
                    });
                }
            })
            .on(&quot;end&quot;, function () {
                d3.select(this).attr(&quot;style&quot;, INTERACTION_STYLE.idleCursor);
                dragTarget = null;
            });

        node.call(drag);

        return svg.node();
    }

    // Expose renderZx both globally (for inline &lt;script&gt; usage) and as a
    // CommonJS-style export for bundlers/tests.

## Detector Error Models

Tsim allows to compute detector error models from a circuit. The method `Circuit.detector_error_model()` computes a `stim.DetectorErrorModel` from the circuit. As opposed to Stim, detectors and observables need not be deterministic.


```python
c = Circuit("""
    RX 0
    R 1
    CNOT 0 1
    DEPOLARIZE1(0.1) 0 1
    M 0 1
    DETECTOR rec[-1] rec[-2]
    DETECTOR rec[-1]
""")
c.detector_error_model()
```




```text title="Output"
stim.DetectorErrorModel('''
    error(0.0666667) D0
    error(0.0666667) D0 D1
    error(0.5) D1
''')
```
