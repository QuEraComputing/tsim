    trees

    Outcome trees: each key's measurement outcomes, expanded as shots reach them.

    A component's amplitudes depend on a shot only through its key and the outcomes of the measurements that split lanes (tables). For one key these outcomes form a binary tree. A node is a measurement whose outcome is random given the outcomes before it, and it holds the chance of outcome 1. An edge holds the output pattern of the branch bits it sets: those of the node's outcome and of every measurement up to the next node whose outcome is certain, so a shot makes no draw there.

    A shot walks its key's tree: at each node it draws its outcome from its own stream, XORs the edge's pattern into its outputs and tests the postselected outputs that the edge makes ready. It stops at the end of the tape or at the first output that discards it. That draws each outcome from its chance given the earlier ones, as lanes do, and needs no sorting of shots.

    The trees are expanded as shots reach them. A shot at a node or key without its edge parks; after a pass over the batch the engine runs the tape on one state per missing edge, up to the next node, and the parked shots resume. The trees persist across batches and calls, and every postselection mask uses them, since they key without postselection. A node's contents depend only on its key and the outcomes before it, and a shot's draws only on its index, so the samples don't depend on the threads, the parking rounds or what was built before.

    Readers take the store's published counts and treat ids beyond them as missing; one writer at a time appends nodes and edges and links them in, then publishes the new counts. Arrays that run out of rows are copied into larger ones, and a reader keeps reading the arrays it took, whose rows below its counts don't change. Links store an id plus one, so that zero means missing.

    def __init__(self, component: StatevectorComponent) -> None:
    """Start with empty trees."""
    self.component = component
    self.layout = layout = sharing_layout(component)
    tape, num_words = component.tape, component.num_words
    num_f = len(tape.f_selection)
    output_bits = _unpack(tape.output_row, num_words)
    key_bits = _unpack(layout.key_rows, num_words)[:, :num_f]
    self.key_words = max(1, (len(key_bits) + 63) // 64)
    self._key_columns = np.zeros((num_f, self.key_words), np.uint64)
    if len(key_bits):
        self._key_columns[:] = _packed(key_bits.T)
    self._output_columns = _packed(output_bits[:, :num_f].T)
    coins = np.flatnonzero(_unpack(layout.shot_dormant, num_words))
    self._coin_columns = _packed(output_bits[:, coins].T)
    output_words = self._output_columns.shape[1]
    lane_bits = _unpack(layout.lane_branches, num_words)
    branch_pattern = np.zeros((len(tape.kind), output_words), np.uint64)
    for a in np.flatnonzero(tape.branch >= 0):
        if lane_bits[tape.branch[a]]:
            column = output_bits[:, tape.branch[a]][None, :]
            branch_pattern[a] = _packed(column)[0]
    self._fixed = _Fixed(
        branch_pattern=branch_pattern,
        lane_dormant=layout.lane_dormant,
        factor=layout.factor,
        coupling=layout.coupling,
        i_power=layout.i_power,
    )
    splitting = np.isin(tape.kind, (MEASURE, MEASURE_DIAG)) | layout.lane_dormant
    self._stride = int(np.count_nonzero(splitting)) + 1
    self.max_slots = max(1, min(2**20, _STATE_BYTES // (16 << component.width)))
    self._arrays = self._empty(num_words, output_words)
    self._counts = np.zeros(5, dtype=np.int64)
    # NOTE: readers take this pair without the lock; the writer replaces it.
    self._published = (self._arrays, self._counts.copy())
    self._lock = threading.Lock()

    def __getstate__(self) -> dict:
    """Return the state to pickle: the used rows of the arrays, no lock."""
    state = self.__dict__.copy()
    del state["_lock"], state["_published"]
    counts = self._counts
    used = {"key_symbols": counts[_KEYS], "states": counts[_SLOTS]}
    used["slot_owner"] = counts[_SLOTS]
    for name in _Arrays._fields:
        if name.startswith("node_"):
            used[name] = counts[_NODES]
        elif name.startswith("edge_"):
            used[name] = counts[_EDGES]
    state["_arrays"] = {
        name: array[: used[name]] if name in used else array
        for name, array in self._arrays._asdict().items()
    }
    return state

    def __setstate__(self, state: dict) -> None:
    """Restore a pickled state, with room to grow and a new lock."""
    used = state.pop("_arrays")
    self.__dict__.update(state)
    output_words = self._output_columns.shape[1]
    arrays = self._empty(self.component.num_words, output_words)
    changes = {}
    for name, array in used.items():
        target = getattr(arrays, name)
        if name.startswith("slot_") and name != "slot_owner":
            changes[name] = array.copy()
            continue
        if len(array) > len(target):
            target = np.empty((2 * len(array), *array.shape[1:]), array.dtype)
        target[: len(array)] = array
        changes[name] = target
    self._arrays = arrays._replace(**changes)
    self._published = (self._arrays, self._counts.copy())
    self._lock = threading.Lock()

    def sample(
    self, f_params: np.ndarray, seed: int, layout: SharingLayout
) -> tuple[np.ndarray, float, dict]:
    """Sample one batch, as ``sample_shared_component`` does with ``layout``.

    ``layout`` gives the outputs' reference bits and the postselected outputs:
    a shot that one of them discards reads ``False`` in the outputs set after
    the discarding output, as if its lane were pruned there.

    Returns:
        ``(samples, max_norm_deviation, stats)``, where ``stats`` counts the
        nodes in total, those this batch built, its parking rounds and its
        parked shots.

    """
    selection = self.component.tape.f_selection
    shot_keys, f_patterns = _shot_keys(
        f_params.shape[0] * (len(selection) + 1),
        f_params,
        selection,
        self._key_columns,
        self._output_columns,
        kernels.PARALLEL_WORK,
        numba.get_num_threads(),
    )

    def symbols_of(rows: np.ndarray) -> np.ndarray:
        return packed_symbols(
            len(rows) * len(selection),
            f_params[rows],
            selection,
            self.component.num_words,
            kernels.PARALLEL_WORK,
            numba.get_num_threads(),
        )

    return self.sample_keyed(shot_keys, f_patterns, symbols_of, seed, layout)

    def sample_keyed(
    self,
    shot_keys: np.ndarray,
    f_patterns: np.ndarray,
    symbols_of: Callable[[np.ndarray], np.ndarray],
    seed: int,
    layout: SharingLayout,
) -> tuple[np.ndarray, float, dict]:
    """Sample one batch from each shot's key and its f-bits' output pattern.

    ``shot_keys`` has shape ``(B, key_words)`` and ``f_patterns`` ``(B,
    words)``; ``symbols_of(rows)`` returns the packed f-bits
    (``packed_symbols``) of the shots ``rows``, which new roots need.
    Otherwise as ``sample``.
    """
    shots = f_patterns.shape[0]
    output_words = self._output_columns.shape[1]
    threads = numba.get_num_threads()
    postselected = layout.prune_first | layout.prune_outputs.any(axis=0)
    checks = np.flatnonzero(postselected)
    checks = checks[np.argsort(layout.ready[checks], kind="stable")]
    draws, coins, _ = _streams(np.uint64(seed))
    at, outcome, depth, pruned, patterns = _start(
        shots * (output_words + len(self._coin_columns) + 1),
        f_patterns,
        self._coin_columns,
        _packed(layout.output_const[None, :])[0],
        checks,
        layout.ready,
        coins,
        kernels.PARALLEL_WORK,
        threads,
    )
    ptr = np.searchsorted(layout.ready[checks], 0).repeat(shots).astype(np.int64)
    active = np.flatnonzero(at != _DONE)
    need = np.full(shots, _NEEDS_NOTHING, dtype=np.int64)
    deviation, built, rounds, parked = 0.0, 0, 0, 0
    while len(active):
        arrays, counts = self._published
        _walk(
            len(active) * self._stride,
            active,
            shot_keys,
            at,
            outcome,
            depth,
            pruned,
            ptr,
            patterns,
            need,
            checks,
            layout.ready,
            arrays.slot_keys,
            arrays.slot_roots,
            arrays.node_threshold,
            arrays.node_child,
            arrays.edge_pattern,
            arrays.edge_next,
            arrays.edge_last,
            counts[_EDGES],
            draws,
            self._stride,
            kernels.PARALLEL_WORK,
            threads,
        )
        active = active[need[active] != _NEEDS_NOTHING]
        if not len(active):
            break
        rounds += 1
        parked += len(active)
        expanded = self._expand(
            arrays,
            active,
            need,
            shot_keys,
            symbols_of,
            (depth, patterns, ptr, checks, layout.ready, draws),
        )
        deviation = _worse(deviation, expanded[0])
        built += expanded[1]
    samples = _outputs(
        shots * self.component.num_outputs, patterns, layout.ready, pruned
    )
    stats = {
        "tree shots": shots,
        "nodes": self.num_nodes,
        "nodes built": built,
        "rounds": rounds,
        "parked": parked,
    }
    return samples, deviation, stats
